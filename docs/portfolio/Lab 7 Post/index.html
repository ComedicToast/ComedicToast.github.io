Function 1 to get all relevant json info

# Function to get JSON info
get_json_info_cleaned <- function(df) {
  # pmap iterates by state, lat and long to get everything we need in a list
  results <- purrr::pmap(df, function(state, latitudes, longitudes) {
    url <- paste0("https://api.g7vrd.co.uk/v1/satellite-passes/25544/", latitudes, "/", longitudes, ".json")
  # get results from the url
    response <- GET(url)
  # if there were no issues
    if (http_status(response)$category == "Success") {
      #grab the json char data
      json_data <- fromJSON(rawToChar(response$content))
      passes_data <- head(json_data$passes$tca, 3)
      #store in a list
      list(state = state, latitude = latitudes, longitude = longitudes, passes = passes_data)
    } else {
      #error message otherwise
      warning(paste0("Error fetching data for ", state, " (lat: ", latitudes, ", long: ", longitudes, "): ", http_status(response)$message))
      list(state = state, latitude = latitudes, longitude = longitudes, error = http_status(response)$message)
    }
  })
  return(results)
}

Function 2 to convert it to a nice df

api_data_cleaned <- get_json_info_cleaned(state_capitals_clean)

# Function to convert API data to a nice df 
convert_api_to_df_cleaned <- function(api_data) {
  # Extract core information using map_dfr for a data frame
  df_core <- purrr::map_dfr(api_data, function(item) {
    data.frame(
      state = item$state,
      latitude = item$latitude,
      longitude = item$longitude
    )
  })

  # grab passes and make sure all have 3, fill with NA if fewer
  all_passes_padded <- purrr::map(api_data, function(item) {
    passes <- item$passes
    if (is.null(passes)) { # Handle cases where there might be an error and no passes
      return(rep(NA_character_, 3))
    }
    length(passes) <- 3 # Pad with NA if less than 3, truncates if more than 3
    return(passes)
  })

  # Convert the list of passes into df by rowbinding with do.call
  passes_df <- as.data.frame(do.call(rbind, all_passes_padded))
  colnames(passes_df) <- paste0("pass", 1:ncol(passes_df))

  # Combine the core data frame with the passes data frame
  final_df <- cbind(df_core, passes_df)

  return(final_df)
}

final_state_capitals_cleaned <- convert_api_to_df_cleaned(api_data_cleaned)

This map does not have a label for Juneau, Alaska; Puerto Rico; and Washington D.C. We removed Alaska since there was no pass times for the ISS sattelite. We removed Puerto Rico since this was a U.S. Territory and we wanted to include the United States capitals. Finally, we removed Washington D.C., since it is not a state capital. This map should reflect only the 50 U.S. State Capitals, with the exception being Juneau as there were no pass times.
